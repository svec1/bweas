import 'base'

t_object: ctemplate = {str = 'clang++:{}.o(NULL) -> objects: -c [source_files:*.cpp] -o OUTPUT_FILE -I[include_paths] [compile_flags]'}
t_lib: ctemplate = {str = 'ar:bweas.a(objects) -> LIBRARY: rcs OUTPUT_FILE {objects}'}
t_exe: ctemplate = {str =' clang++:{}(bweas_slib) -> EXECUTABLE: INPUT_FILE {bweas_slib} -o OUTPUT_FILE -I[include_paths] -l[libs] [compile_flags]'}

lib_names: string[] = ['lz4', 'lua', 'luajit-5.1']
target_compile_flags: string[] = ['-std=gnu++23', '-Wall', '-fPIC']
target_include_paths: string[] = ['/usr/include', '/usr/local/include', config_path+'/src/bs']

bweas_slib: target = {
                        extension = cxx_lib,
                        templates = ['t_object', 't_lib'],
                        compile_flags = target_compile_flags,
                        include_paths = target_include_paths,
                        source_files = file(get_files, 'src/bs/*.cpp'),
                        threads = 6 
}

bweas_exe: target = {
                    extension = cxx_exec,
                    templates = ['t_exe'],
                    dependencies = ['bweas_slib'],
                    libs = lib_names,
                    compile_flags = target_compile_flags,
                    include_paths = target_include_paths,
                    source_files = file(get_files, 'src/bweas.cpp')
}

build(bweas_slib, bweas_exe)
