import "base"

set(DECL_CONFIG_VAR, 1)

create_template(t_flex,  "flex:lexer.cpp(NULL) -> [source_files]: <'-o'> OUTPUT_FILE <[source_files:lexer.l]>")
create_template(t_bison, "bison:parser.cpp(NULL) -> [source_files]: <'-o'> OUTPUT_FILE <'-H'> <[source_files:parser.y]>")
create_template(t_obj,   "clang++:{}.o(NULL) -> OBJ_S: <'-c'> <[source_files:*.cpp]> <'-o'> OUTPUT_FILE <'-Wall'> <'-std=gnu++23'> <'-fPIC'> <'-I'[include_paths]>")
create_template(t_lib,   "ar:bweas.a(OBJ_S) -> LIBRARY: <'rcs'> OUTPUT_FILE <{OBJ_S}>")

create_template(t_exec,  "clang++:bweas(BWEAS_SLIB) -> EXECUTABLE: INPUT_FILE <{BWEAS_SLIB}> <'-o'> OUTPUT_FILE <'-I'[include_paths]> <'-std=gnu++23'> <'-llz4'> <'-llua'> <'-lluajit-5.1'>")

file(SRC_FILES, 0, "src/bs/*.cpp", "src/bs/utils/*.cpp", "src/bs/lang/*.cpp", "src/bs/lang/lexer.l", "src/bs/lang/parser.y")
file(BWEAS_MAIN_FILE, 0, "src/bweas.cpp")

create_target(BWEAS_SLIB, CPP_LIBRARY, {LIBRARY}, {SRC_FILES})
create_target(BWEAS, CPP_EXECUTABLE, {EXECUTABLE}, {BWEAS_MAIN_FILE})

add_dependencies_target(BWEAS, BWEAS_SLIB)

set(BWEAS_SLIB_include_paths, "../src/bs/")
set(BWEAS_include_paths, "../src/bs/")

use_templates(BWEAS_SLIB, t_flex, t_bison, t_obj, t_lib)
use_templates(BWEAS, t_exec)
